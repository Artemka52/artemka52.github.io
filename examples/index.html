Практические примеры

1. Debounce

Откладывает вызов функции, пока не пройдёт пауза без новых вызовов.

function debounce(fn, delay = 300) {
  let timerId;
  return function (...args) {
    clearTimeout(timerId);
    timerId = setTimeout(() => fn.apply(this, args), delay);
  };
}

const search = debounce((query) => {
  console.log('Поиск:', query);
}, 500);

2. Throttle

Ограничивает частоту вызовов - не чаще раза в N мс.

function throttle(fn, limit = 200) {
  let lastCall = 0;
  return function (...args) {
    const now = Date.now();
    if (now - lastCall >= limit) {
      lastCall = now;
      fn.apply(this, args);
    }
  };
}

window.addEventListener('scroll', throttle(() => {
  console.log('scroll:', window.scrollY);
}, 100));

3. Глубокое клонирование

function deepClone(obj, seen = new WeakMap()) {
  if (obj === null || typeof obj !== 'object') return obj;
  if (seen.has(obj)) return seen.get(obj);

  const clone = Array.isArray(obj) ? [] : {};
  seen.set(obj, clone);

  for (const key of Reflect.ownKeys(obj)) {
    clone[key] = deepClone(obj[key], seen);
  }
  return clone;
}

4. Группировка массива

function groupBy(arr, keyFn) {
  return arr.reduce((acc, item) => {
    const key = keyFn(item);
    (acc[key] ||= []).push(item);
    return acc;
  }, {});
}

const users = [
  { name: 'Alex', role: 'admin' },
  { name: 'Bob',  role: 'user' },
  { name: 'Ann',  role: 'admin' },
];

console.log(groupBy(users, u => u.role));
// { admin: [Alex, Ann], user: [Bob] }

7. Async-очередь с ограничением

class TaskQueue {
  constructor(limit = 2) {
    this.limit = limit;
    this.running = 0;
    this.queue = [];
  }

  add(task) {
    return new Promise((resolve, reject) => {
      this.queue.push({ task, resolve, reject });
      this.run();
    });
  }

  async run() {
    if (this.running >= this.limit || !this.queue.length) return;

    const { task, resolve, reject } = this.queue.shift();
    this.running++;

    try {
      resolve(await task());
    } catch (err) {
      reject(err);
    } finally {
      this.running--;
      this.run();
    }
  }
}
