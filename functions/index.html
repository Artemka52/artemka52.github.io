Функции

Способы объявления

// Function Declaration - поднимается
function add(a, b) {
  return a + b;
}

// Function Expression
const sub = function (a, b) {
  return a - b;
};

// Стрелочная функция
const mul = (a, b) => a * b;

// Стрелочная с блоком
const div = (a, b) => {
  if (b === 0) throw new Error('Division by zero');
  return a / b;
};

Параметры по умолчанию

function greet(name = 'гость', greeting = 'Привет') {
  return `${greeting}, ${name}!`;
}

greet();                  // 'Привет, гость!'
greet('Alex');            // 'Привет, Alex!'
greet('Alex', 'Здорово'); // 'Здорово, Alex!'

Rest-параметры

function sum(...nums) {
  return nums.reduce((s, n) => s + n, 0);
}
sum(1, 2, 3, 4, 5); // 15

Стрелочные vs обычные

  • this - у обычной динамический, у стрелочной лексический.
  • arguments - есть у обычной, нет у стрелочной.
  • new - работает с обычной, не работает со стрелочной.
  • Hoisting - есть у function declaration, нет у стрелочных.

Функции высшего порядка

const nums = [1, 2, 3, 4, 5];

const doubled = nums.map(n => n * 2);    // [2, 4, 6, 8, 10]
const evens   = nums.filter(n => n % 2 === 0); // [2, 4]
const total   = nums.reduce((acc, n) => acc + n, 0); // 15

Рекурсия

function factorial(n) {
  if (n <= 1) return 1;
  return n * factorial(n - 1);
}

factorial(5); // 120
