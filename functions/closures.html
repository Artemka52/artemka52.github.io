Замыкания (Closures)

Замыкание - функция, «запоминающая» переменные из внешней области видимости, даже если внешняя функция уже завершилась.

Простой пример

function makeCounter() {
  let count = 0;

  return function () {
    count++;
    return count;
  };
}

const counter = makeCounter();
console.log(counter()); // 1
console.log(counter()); // 2
console.log(counter()); // 3

Приватное состояние

function createBankAccount(initial) {
  let balance = initial;

  return {
    deposit: (amount) => { balance += amount; },
    withdraw: (amount) => {
      if (amount > balance) throw new Error('Недостаточно средств');
      balance -= amount;
    },
    getBalance: () => balance,
  };
}

const acc = createBankAccount(100);
acc.deposit(50);
acc.withdraw(30);
console.log(acc.getBalance()); // 120
console.log(acc.balance);      // undefined

Фабрики функций

function multiplier(factor) {
  return (n) => n * factor;
}

const double = multiplier(2);
const triple = multiplier(3);

console.log(double(5));  // 10
console.log(triple(5));  // 15

Классическая ошибка с var в цикле

// ПЛОХО: var — общая переменная
for (var i = 0; i < 3; i++) {
  setTimeout(() => console.log(i), 0);
}
// 3, 3, 3

// ХОРОШО: let - новая привязка на каждой итерации
for (let i = 0; i < 3; i++) {
  setTimeout(() => console.log(i), 0);
}
// 0, 1, 2

IIFE - Immediately Invoked Function Expression

const module = (function () {
  const secret = 'hidden';
  return {
    reveal: () => secret,
  };
})();

console.log(module.reveal()); // 'hidden'
