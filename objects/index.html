Объекты и прототипы

Создание объектов

// Литерал
const user = { name: 'Alex', age: 30 };

// Конструктор
const user2 = new Object();
user2.name = 'Bob';

// Object.create
const proto = { greet() { return 'hi'; } };
const obj = Object.create(proto);

Доступ к свойствам

const user = { name: 'Alex', 'favorite-color': 'blue' };

user.name;                 // 'Alex'
user['name'];              // 'Alex'
user['favorite-color'];    // 'blue'
user.city;                 // undefined

Деструктуризация

const { name, age = 18, ...rest } = user;

Методы объекта

const calc = {
  value: 0,
  add(n) { this.value += n; return this; },
  sub(n) { this.value -= n; return this; },
};

calc.add(5).sub(2);
console.log(calc.value); // 3

Геттеры и сеттеры

const person = {
  firstName: 'Alex',
  lastName: 'Ivanov',
  get fullName() {
    return `${this.firstName} ${this.lastName}`;
  },
  set fullName(value) {
    [this.firstName, this.lastName] = value.split(' ');
  },
};

person.fullName = 'Maria Petrova';
console.log(person.firstName); // 'Maria'

Прототипы

const animal = {
  speak() { return `${this.name} издаёт звук`; },
};

const dog = Object.create(animal);
dog.name = 'Rex';

console.log(dog.speak()); // 'Rex издаёт звук'

Классы

class Animal {
  constructor(name) { this.name = name; }
  speak() { return `${this.name} издаёт звук`; }
}

class Dog extends Animal {
  constructor(name, breed) {
    super(name);
    this.breed = breed;
  }
  speak() { return `${this.name} (${this.breed}) лает`; }
}

const rex = new Dog('Rex', 'Labrador');
console.log(rex.speak());           // 'Rex (Labrador) лает'
console.log(rex instanceof Dog);    // true
console.log(rex instanceof Animal); // true
