async / await

Синтаксический сахар над промисами. Позволяет писать асинхронный код «как синхронный».

Основы

async function fetchUser(id) {
  const res = await fetch(`/api/users/${id}`);
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return res.json();
}

const user = await fetchUser(1);
console.log(user);

await работает только внутри async-функции или на верхнем уровне модуля.

Возвращаемое значение

async function answer() { return 42; }

answer().then(v => console.log(v)); // 42

Обработка ошибок

async function loadData() {
  try {
    const res = await fetch('/api/data');
    return await res.json();
  } catch (err) {
    console.error('Ошибка:', err);
    throw err;
  } finally {
    console.log('Загрузка завершена');
  }
}

Последовательно vs параллельно

// Последовательно — медленно
const user  = await fetch('/api/user').then(r => r.json());
const posts = await fetch('/api/posts').then(r => r.json());

// Параллельно — быстро
const [user2, posts2] = await Promise.all([
  fetch('/api/user').then(r => r.json()),
  fetch('/api/posts').then(r => r.json()),
]);

Циклы с await

// Последовательно
for (const id of [1, 2, 3]) {
  const user = await fetchUser(id);
  console.log(user);
}

// Параллельно
await Promise.all([1, 2, 3].map(id => fetchUser(id)));

Не используйте await внутри forEach — он не дожидается результата.

Практический пример: retry

async function fetchWithRetry(url, attempts = 3) {
  for (let i = 0; i < attempts; i++) {
    try {
      const res = await fetch(url);
      if (res.ok) return res.json();
    } catch (err) {
      if (i === attempts - 1) throw err;
      await new Promise(r => setTimeout(r, 1000 * (i + 1)));
    }
  }
}
